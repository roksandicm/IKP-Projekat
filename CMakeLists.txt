cmake_minimum_required(VERSION 3.20)
project(IKPProjekat LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This project uses Windows threads and Winsock.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ikp_common src/protocol.cpp src/network.cpp src/topology.cpp)
target_include_directories(ikp_common PUBLIC include)
target_compile_definitions(ikp_common PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0600)
target_link_libraries(ikp_common PUBLIC ws2_32)
add_library(ikp_student1 src/transport.cpp src/routing.cpp)
target_link_libraries(ikp_student1 PUBLIC ikp_common)
add_executable(data_source src/data_source.cpp)
target_link_libraries(data_source PRIVATE ikp_common)
add_executable(student1_tests tests/student1_tests.cpp)
target_link_libraries(student1_tests PRIVATE ikp_student1)
foreach(target ikp_common ikp_student1 data_source student1_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
enable_testing()
add_test(NAME student1 COMMAND student1_tests "${CMAKE_CURRENT_SOURCE_DIR}/config/example.topology" "$<TARGET_FILE:data_source>")
set_tests_properties(student1 PROPERTIES TIMEOUT 120)
